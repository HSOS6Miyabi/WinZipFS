cmake_minimum_required(VERSION 3.24)
project(WinZipFS VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "WinZipFS requires Windows and MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(bit7z CONFIG REQUIRED)
find_package(7zip CONFIG REQUIRED)
find_package(libzip CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
set(WINFSP_ROOT "C:/Program Files (x86)/WinFsp" CACHE PATH "WinFsp installation")
if(CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
    set(WINFSP_ARCH a64)
elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(WINFSP_ARCH x64)
else()
    set(WINFSP_ARCH x86)
endif()
find_library(WINFSP_LIB NAMES winfsp-${WINFSP_ARCH} PATHS "${WINFSP_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
add_library(WinZipFSCore SHARED src/session.cpp)
target_include_directories(WinZipFSCore PRIVATE third_party "${WINFSP_ROOT}/inc" $<TARGET_PROPERTY:7zip::7zip,INTERFACE_INCLUDE_DIRECTORIES>)
target_compile_definitions(WinZipFSCore PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(WinZipFSCore PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(WinZipFSCore PRIVATE bit7z::bit7z libzip::zip ZLIB::ZLIB "${WINFSP_LIB}" delayimp ole32 shell32 advapi32 bcrypt)
target_link_options(WinZipFSCore PRIVATE /DELAYLOAD:winfsp-${WINFSP_ARCH}.dll)
enable_testing()
add_executable(core-tests tests/core.cpp)
target_link_libraries(core-tests PRIVATE WinZipFSCore)
target_compile_definitions(core-tests PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(core-tests PRIVATE /W4 /utf-8 /EHsc)
add_test(NAME native-core COMMAND core-tests)
